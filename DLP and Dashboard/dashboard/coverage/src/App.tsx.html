import {useEffect,useState} from 'react';
import {getMetrics} from './api'; import {connectEvents} from './websocket'; import type {Alert,Event,Metric} from './types'; import {Metrics} from './components/Metrics'; import {Alerts} from './components/Alerts'; import {EventsTable} from './components/EventsTable'; import './styles.css';
export default function App(){const [tab,setTab]=useState<'overview'|'events'>('overview');const [metrics,setMetrics]=useState<Metric[]>([]);const [events,setEvents]=useState<Event[]>([]);const [alerts,setAlerts]=useState<Alert[]>([]);const [status,setStatus]=useState('connecting');
useEffect(()=>{const load=()=>getMetrics().then(setMetrics).catch(()=>setMetrics([]));load();const timer=setInterval(load,5000);const stop=connectEvents(e=>{setEvents(v=>[e,...v].slice(0,50));if(e.event_type==='validation'&&e.allowed===false)setAlerts(v=>[{id:crypto.randomUUID(),severity:'high',message:`Blocked entities: ${(e.blocked_entities||[]).join(', ')}`,timestamp:e.timestamp},...v].slice(0,20))},setStatus);return()=>{clearInterval(timer);stop()}},[]);
return <main><header><div><h1>LLM-Guard</h1><p>Data Loss Prevention & Security Monitoring</p></div><span className={`status ${status}`}>{status}</span></header><nav><button onClick={()=>setTab('overview')} className={tab==='overview'?'active':''}>Overview</button><button onClick={()=>setTab('events')} className={tab==='events'?'active':''}>Events</button></nav>{tab==='overview'?<><Metrics metrics={metrics}/><Alerts alerts={alerts}/></>:<EventsTable events={events}/>}</main>}
